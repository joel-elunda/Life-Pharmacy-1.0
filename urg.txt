package life.pharmacy.controllers;

import javafx.fxml.FXML;
import javafx.stage.Stage;
import life.pharmacy.services.ExportImportService;

public class DashboardController {

    private ExportImportService exportImportService = new ExportImportService();

    private Stage getStage() {
        return (Stage) rootPane.getScene().getWindow();
    }

    // Produits
    @FXML private void onExportProduits() {
        exportImportService.exportProduits(getStage());
    }
    @FXML private void onImportProduits() {
        exportImportService.importProduits(getStage());
    }

    // Employés
    @FXML private void onExportEmployes() {
        exportImportService.exportEmployes(getStage());
    }
    @FXML private void onImportEmployes() {
        exportImportService.importEmployes(getStage());
    }

    // Clients
    @FXML private void onExportClients() {
        exportImportService.exportClients(getStage());
    }
    @FXML private void onImportClients() {
        exportImportService.importClients(getStage());
    }

    // Fournisseurs
    @FXML private void onExportFournisseurs() {
        exportImportService.exportFournisseurs(getStage());
    }
    @FXML private void onImportFournisseurs() {
        exportImportService.importFournisseurs(getStage());
    }

    // Factures
    @FXML private void onExportFactures() {
        exportImportService.exportFactures(getStage());
    }
    @FXML private void onImportFactures() {
        exportImportService.importFactures(getStage());
    }

    // Transactions
    @FXML private void onExportTransactions() {
        exportImportService.exportTransactions(getStage());
    }
    @FXML private void onImportTransactions() {
        exportImportService.importTransactions(getStage());
    }

    // Recettes
    @FXML private void onExportRecettes() {
        exportImportService.exportRecettes(getStage());
    }
    @FXML private void onImportRecettes() {
        exportImportService.importRecettes(getStage());
    }
}
